مشروع 24: لعبة الكلمة المخفيّة (Hangman) ◆ مشروع كبير

الهدف: أشهر مشروع تدريبي في البرمجة — قوائم + حلقات + شروط + عشوائية.

1. المنطق قبل الكود

  1) اختار كلمة عشوائية من قائمة
  2) اعمل قائمة عرض بعدد حروف الكلمة، كلها مخفيّة  ["_","_","_"]
  3) كرّر:
       اسأل المستخدم على حرف
       لو الحرف في الكلمة → اكشفه في كل مواضعه
       لو مش موجود      → انقص محاولة
       اعرض حالة العرض
       لو مفيش مخفي     → كسب، اخرج
       لو المحاولات خلصت → خسر، اخرج
خوارزمية اللعبة

2. النسخة الأساسية

import random
words = ["python", "programming", "challenge", "life", "code", "man", "blind", "love", "hangman"]
word_random = random.choice(words)
display = [" " for _ in word_random]
print("".join(display))
attempts = 6
while attempts > 0:
    user_letter = input("Enter your guessed letter: ").lower()
    for letter in range(len(word_random)):
        if word_random[letter] == user_letter:
            display[letter] = user_letter
    if user_letter not in word_random:
        attempts -= 1
        print(f"Letter {user_letter} not found. Remaining attempts: {attempts}")
    print("".join(display))
    if " " not in display:
        print(f"Congratulations! You guessed the word: {''.join(display)}")
        break
    if attempts == 0:
        print(f"Sorry, you ran out of attempts. The correct word was: {word_random}")
        break

💥 عيوب في النسخة الأساسية
  1. لو المستخدم كرّر نفس الحرف الغلط، بيخسر محاولة تانية.
  2. لو دخل كلمة كاملة بدل حرف، البرنامج بيتصرّف بغرابة.
  3. مفيش عرض للحروف اللي اتجرّبت.
  4. المسافات في العرض صعبة القراية — الشرطة السفلية أوضح.

4. النسخة الاحترافية

import random
WORDS = ["python", "programming", "challenge", "developer", "function",
         "variable", "algorithm", "computer", "keyboard", "internet"]
MAX_ATTEMPTS = 6
def choose_word():
    return random.choice(WORDS)
def get_display(word, guessed):
    """بترجّع نص العرض: الحروف المكتشفة والباقي شرطات"""
    return " ".join(c if c in guessed else "_" for c in word)
def draw_hangman(wrong):
    """بترسم حالة اللعبة"""
    stages = [
        "  +---+\n      |\n      |\n      |\n     ===",
        "  +---+\n  O   |\n      |\n      |\n     ===",
        "  +---+\n  O   |\n  |   |\n      |\n     ===",
        "  +---+\n  O   |\n /|   |\n      |\n     ===",
        "  +---+\n  O   |\n /|\\  |\n      |\n     ===",
        "  +---+\n  O   |\n /|\\  |\n /    |\n     ===",
        "  +---+\n  O   |\n /|\\  |\n / \\  |\n     ===",
    ]
    print(stages[wrong])
def play_round():
    word = choose_word()
    guessed = set()
    wrong = 0
    print("\n" + "=" * 34)
    print(f"الكلمة فيها {len(word)} حرف")
    print("=" * 34)
    while wrong < MAX_ATTEMPTS:
        draw_hangman(wrong)
        print(f"\n  {get_display(word, guessed)}")
        print(f"  حروف مجرّبة: {' '.join(sorted(guessed)) or '(لا شيء)'}")
        print(f"  محاولات متبقية: {MAX_ATTEMPTS - wrong}")
        letter = input("\n  خمّن حرف: ").strip().lower()
        # التحقّق من المدخل
        if len(letter) != 1 or not letter.isalpha():
            print("  ⚠️ اكتب حرف واحد بس")
            continue
        if letter in guessed:
            print("  ⚠️ جرّبت الحرف ده قبل كده")
            continue
        guessed.add(letter)
        if letter in word:
            print(f"  ✅ صح! الحرف '{letter}' موجود")
            if all(c in guessed for c in word):
                print(f"\n  🎉 مبروك! الكلمة هي: {word}")
                return True
        else:
            wrong += 1
            print(f"  ❌ الحرف '{letter}' مش موجود")
    draw_hangman(wrong)
    print(f"\n  💀 خسرت. الكلمة كانت: {word}")
    return False
# --- التشغيل ---
print("لعبة الكلمة المخفيّة")
wins = losses = 0
while True:
    if play_round():
        wins += 1
    else:
        losses += 1
    print(f"\n  النتيجة: {wins} فوز - {losses} خسارة")
    if input("\n  تلعب تاني؟ (yes/no) ").strip().lower() == "no":
        print("  شكرًا للعب!")
        break

مشروع 25: حجر ورقة مقص ◆ مشروع

الهدف: منطق شروط مركّب + إدارة محاولات + قائمة مساعدة.

import random
greeting = input("مرحبا بك في لعبة: حجر ورقة مقص, اضغط إدخال للبدء, أو 'مساعدة' لعرض المساعدة\n")
help_text = ("في هذه اللعبة يمكنك أن تختار بين ثلاثة عناصر: الحجر والورقة والمقص, "
             "والكمبيوتر أيضا سيختار واحدة, وسيكون منكما فائز بحسب اختياركما, بالتوفيق")
if greeting == "مساعدة" or greeting == "المساعدة":
    print(help_text)
choices = ["حجر", "ورقة", "مقص"]
computer_choice = random.choice(choices)
attempts = 4
while attempts > 0:
    user_choice = input("من فضلك اختر بين العناصر: حجر, ورقة, مقص: ").strip()
    if (user_choice == "حجر" and computer_choice == "مقص" or
            user_choice == "مقص" and computer_choice == "ورقة" or
            user_choice == "ورقة" and computer_choice == "حجر"):

        attempts -= 1
        print(f"حظا أوفر: فاز الكمبيوتر, اختار {computer_choice} وإنت اخترت {user_choice}. "

    if attempts == 0:
        print("لقد استنفدت جميع محاولاتك, لقد خسرت")
        break
    if user_choice == "إغلاق":
        print("شكرا لاختيار هذه اللعبة, نراك قريبا")
        break
print("شكرا لك لاختيار لعبتنا, نرجو لك التوفيق دائما")

import random
CHOICES = ["حجر", "ورقة", "مقص"]
BEATS = {"حجر": "مقص", "مقص": "ورقة", "ورقة": "حجر"}   # المفتاح يكسب القيمة
HELP_TEXT = "اختر حجر أو ورقة أو مقص. الحجر يكسر المقص، المقص يقص الورقة، الورقة تغلّف الحجر."
MAX_LOSSES = 4
start = input("أهلًا بك في حجر ورقة مقص. اضغط Enter للبدء أو اكتب 'مساعدة': ").strip()
if start in ["مساعدة", "المساعدة", "help"]:
    print(HELP_TEXT)
losses = 0
wins = 0
draws = 0
while losses < MAX_LOSSES:
    user = input(f"\nاختيارك ({' / '.join(CHOICES)}) أو 'إغلاق': ").strip()
    if user == "إغلاق":
        print("شكرًا لك، نراك قريبًا")
        break
    if user not in CHOICES:
        print("⚠️ اختيار غير صحيح، جرّب تاني")
        continue
    computer = random.choice(CHOICES)          # ✅ اختيار جديد كل جولة
    print(f"الكمبيوتر اختار: {computer}")
    if user == computer:
        draws += 1
        print("🤝 تعادل")
    elif BEATS[user] == computer:
        wins += 1
        print(f"🎉 كسبت! {user} تغلب {computer}")
    else:
        losses += 1
        print(f"❌ خسرت. {computer} تغلب {user}. متبقّي لك {MAX_LOSSES - losses}")
if losses == MAX_LOSSES:
    print("\n💀 استنفدت محاولاتك")
print(f"\nالنتيجة النهائية: {wins} فوز — {losses} خسارة — {draws} تعادل")

مشروع 26: مدير المهام الكامل (Task Manager) ◆ المشروع النهائي

الهدف: أكبر مشروع في الكورس — كل حاجة اتعلّمتها في تطبيق واحد كامل.

1. النسخة من المحاضرة

def add_task(tasks):
    name = input("Enter task name: ")
    description = input("Enter task description: ")
    status_input = input("Is the task completed? (true/false): ").lower()
    status = True if status_input == "true" else False
    tasks.append({"name": name, "description": description, "status": status})
    print("Task added successfully!")
def modify_task(tasks):
    if not tasks:
        print("No tasks available.")
        return
    task_name = input("Enter the name of the task to modify: ")
    for task in tasks:
        if task["name"].lower() == task_name.lower():
            option = input("What do you want to modify? (status/description/both): ").lower()
            if option == "status":
                new_status = input("Enter new status (true/false): ").lower() == "true"
                task["status"] = new_status
            elif option == "description":
                new_description = input("Enter new description: ")
                task["description"] = new_description
            elif option == "both":
                new_status = input("Enter new status (true/false): ").lower() == "true"
                new_description = input("Enter new description: ")
                task["status"] = new_status
                task["description"] = new_description
            else:
                print("Invalid option.")
            print("Task modified successfully!")
            return
    print("Task not found.")
def delete_task(tasks):
    if not tasks:
        print("No tasks available.")
        return
    task_name = input("Enter the name of the task to delete: ")
    for i, task in enumerate(tasks):
        if task["name"].lower() == task_name.lower():
            confirm = input("Are you sure you want to delete this task? (yes/no): ").lower()
            if confirm == "yes":
                del tasks[i]
                print("Task deleted successfully!")
                return
            else:
                print("Deletion cancelled.")
                return
    print("Task not found.")
def view_tasks(tasks):
    if not tasks:
        print("No tasks available.")
        return
    completed = [task for task in tasks if task["status"]]
    not_completed = [task for task in tasks if not task["status"]]
    print("Completed tasks:")
    for task in completed:
        print(f"{task['name']}: {task['description']}")
    print("\nNot completed tasks:")
    for task in not_completed:
        print(f"{task['name']}: {task['description']}")
def show_help():
    print("Task Manager Help:")
    print("1: Add new task")
    print("2: Modify task")
    print("3: Delete task")
    print("4: View tasks")
    print("5: Help")
    print("6: Exit")
def main():
    tasks = []
    print("Welcome to Task Manager!")
    while True:
        option = input("Choose an option (1-6): ")
        if option == "1":
            add_task(tasks)
        elif option == "2":
            modify_task(tasks)
        elif option == "3":
            delete_task(tasks)
        elif option == "4":
            view_tasks(tasks)
        elif option == "5":
            show_help()
        elif option == "6":
            print("Exiting Task Manager. Goodbye!")
            break
        else:
            print("Invalid option. Please choose between 1 and 6.")
if __name__ == "__main__":
    main()

بنية البرنامج: دالة واحدة لكل مسؤولية
🧠 التقنيات المهمّة في الكود ده
  • tasks بتتمرّر لكل دالة: بدل ما تكون global. وبما إن القائمة mutable (درس الذاكرة!)، أي تعديل جوّه الدالة بيأثّر على القائمة الأصلية. ده بالظبط اللي إحنا عايزينه هنا.
  • return المبكّر: if not tasks: return — بتخرج من الدالة فورًا لو مفيش مهام، فمش محتاجين else ولا مسافات بادئة زيادة.
  • status = True if x == "true" else False: الشرط المختصر. ملاحظة: ممكن تختصرها أكتر لـ status = (x == "true") لأن المقارنة أصلًا بترجّع bool.
  • التصفية بـ list comprehension: [t for t in tasks if t["status"]] بتقسم المهام لمجموعتين.
  • if __name__ == "__main__": — شوف الصندوق تحت.
ℹ️ إيه هي if __name__ == "__main__":؟

بايثون بتحط في كل ملف متغيّر خفي اسمه __name__. لو الملف اتشغّل مباشرةً، قيمته بتبقى "__main__". لو الملف اتستورد من ملف تاني، قيمته بتبقى اسم الملف.

فالسطر ده معناه: «شغّل main() بس لو الملف ده هو اللي اتشغّل مباشرة». الفايدة: لو حد عمل import task_manager عشان يستخدم add_task بس، البرنامج مش هيبدأ يشتغل عنده تلقائيًا. ده اصطلاح احترافي هتشوفه في كل مشروع بايثون.

import json
import os
from datetime import datetime
DATA_FILE = "tasks.json"
# ---------- طبقة التخزين ----------
def load_tasks():
    """بتقرا المهام من الملف، وترجّع قائمة فاضية لو الملف مش موجود"""
    if not os.path.exists(DATA_FILE):
        return []
    try:
        with open(DATA_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except (json.JSONDecodeError, OSError):
        print("⚠️ الملف تالف، هنبدأ بقائمة فاضية")
        return []
def save_tasks(tasks):
    """بتحفظ المهام في الملف"""
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(tasks, f, ensure_ascii=False, indent=2)
# ---------- طبقة العمليات ----------
def add_task(tasks):
    name = input("اسم المهمّة: ").strip()
    if not name:
        print("❌ الاسم ما ينفعش يكون فاضي")
        return
    description = input("الوصف: ").strip()
    priority = input("الأولوية (عالية/متوسطة/منخفضة) [متوسطة]: ").strip() or "متوسطة"
    tasks.append({
        "name": name,
        "description": description,
        "status": False,
        "priority": priority,

def pick_task(tasks):
    """بتعرض المهام وترجّع فهرس المهمّة اللي اختارها المستخدم، أو None"""
    if not tasks:
        print("مفيش مهام")
        return None

        mark = "✔" if t["status"] else "○"
        print(f"  {i}. [{mark}] {t['name']}")
    try:
        index = int(input("رقم المهمّة: ")) - 1
    except ValueError:
        print("❌ اكتب رقم")
        return None
    if 0 <= index < len(tasks):
        return index
    print("❌ رقم خارج النطاق")
    return None
def modify_task(tasks):

    print("✅ اتعدّلت")
def delete_task(tasks):

    if index is None:

    name = tasks[index]["name"]

def view_tasks(tasks):

    bar = "█" * (percent // 5) + "░" * (20 - percent // 5)

🧠 الدوال ككائنات (First-Class Functions)

actions = {"1": add_task} — لاحظ إن add_task مكتوبة من غير أقواس. يعني إحنا بنخزّن الدالة نفسها، مش نتيجتها. وبعدين actions["1"](tasks) بتضيف الأقواس وتناديها.

في بايثون الدوال قيم زي أي قيمة: تقدر تخزّنها في متغيّر، تحطّها في قائمة، تمرّرها لدالة تانية. ده أساس مفاهيم متقدّمة زي الـ decorators و callbacks، وهيبقى نقطة انطلاقك في التراك الجاي.

افصل «منطق اللعبة» في دوال صغيرة تتختبر لوحدها — ده اللي بيخلّي المشاريع الكبيرة قابلة للتطوير.

🧪 التحدّي 1 — قلب لعبة Hangman ⭐⭐

mask(word, guessed): ترجّع الكلمة بحروفها المخمّنة وشرطة مكان الباقي، مفصولين بمسافة ("p _ _ _ o _")، والمقارنة من غير فرق بين الكبير والصغير (والحرف يظهر بشكله الأصلي).
is_won(word, guessed): True لو كل حروف الكلمة اتخمّنت.

شوف الحل النموذجي 
def mask(word, guessed):
    lowered = {g.lower() for g in guessed}
    return " ".join(ch if ch.lower() in lowered else "_" for ch in word)
def is_won(word, guessed):
    lowered = {g.lower() for g in guessed}
    return set(word.lower()) <= lowered
🧪 التحدّي 2 — حَكَم حجر ورقة مقص ⭐⭐

winner(p1, p2) — الحركات "rock" و "paper" و "scissors". ترجّع "p1" أو "p2" أو "draw"، وترمي ValueError لأي حركة تانية.

شوف الحل النموذجي 
BEATS = {"rock": "scissors", "scissors": "paper", "paper": "rock"}
def winner(p1, p2):
    if p1 not in BEATS or p2 not in BEATS:
        raise ValueError(f"حركة غير معروفة: {p1!r} / {p2!r}")
    if p1 == p2:
        return "draw"
    return "p1" if BEATS[p1] == p2 else "p2"
