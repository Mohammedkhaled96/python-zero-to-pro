الاختبارات: إزاي تتأكّد إن كودك شغّال فعلًا

لحد دلوقتي إحنا بنتأكّد من الكود بـ «شغّله وشوف». الطريقة دي بتنفع لـ 20 سطر، وبتفشل تمامًا في 500. الوحدة دي هي الخط الفاصل بين الهاوي والمحترف: هتكتب كود بيختبر كودك، فأي تعديل بعد كده تعرف في ثانية إنه كسر حاجة ولا لأ.

ليه الاختبارات؟ والمشكلة اللي بتحلّها درس 85

الهدف: تفهم إن الاختبار مش «رفاهية للشركات الكبيرة» — هو أسرع طريقة تشتغل.

المشكلةالشرح

4. pytest.raises: الطريقة النضيفة لاختبار الأخطاء

↩️ المثال ده بيكمّل على مثال قبله في نفس الدرس (بيستخدم divide) — لو هتشغّله جوّه الصفحة، شغّل اللي قبله الأول.

# الطريقة اليدوية (من الدرس السابق) — 5 سطور
try:
    divide(10, 0)
    assert False, "المفروض ترمي خطأ"
except ValueError:
    pass
# مع pytest — سطرين وأوضح
with pytest.raises(ValueError):
    divide(10, 0)
# وبفحص الرسالة كمان
with pytest.raises(ValueError, match="صفر"):
    divide(10, 0)
# وبالوصول لكائن الخطأ
with pytest.raises(ValueError) as info:
    divide(10, 0)
assert "ممنوعة" in str(info.value)

with هنا نفس with بتاعة الملفات — مدير سياق. معناها «الكود اللي جوّايا لازم يرمي الخطأ ده، وإلا الاختبار يفشل».

import re
EMAIL_RE = re.compile(r"^[\w.+-]+@[\w-]+\.[\w.]+$")
def is_valid_email(email):
    """بترجّع True لو الإيميل صالح شكليًا.
    >>> is_valid_email("a@b.com")
    True
    >>> is_valid_email("no-at-sign")
    False
    """
    if not isinstance(email, str):
        raise TypeError("الإيميل لازم يكون نص")
    return bool(EMAIL_RE.match(email.strip()))
def is_valid_egyptian_phone(phone):
    """رقم مصري: 11 خانة، يبدأ بـ 010 أو 011 أو 012 أو 015."""
    if not isinstance(phone, str):
        raise TypeError("الرقم لازم يكون نص")
    phone = phone.strip()
    valid_prefixes = ("010", "011", "012", "015")
    return (phone.isdigit()
            and len(phone) == 11
            and phone.startswith(valid_prefixes))
def password_strength(password):
    """بترجّع درجة من 0 لـ 4."""
    if not isinstance(password, str):
        raise TypeError("كلمة السر لازم تكون نص")
    score = 0
    if len(password) >= 8:
        score += 1
    if any(c.isdigit() for c in password):
        score += 1
    if any(c.isupper() for c in password):
        score += 1
    if any(not c.isalnum() for c in password):
        score += 1
    return score

class Cart:
    def __init__(self):
        self.items = {}
    def add(self, name, price, qty=1):
        if price < 0:
            raise ValueError("السعر ما ينفعش يكون سالب")
        if qty <= 0:
            raise ValueError("الكمية لازم تكون موجبة")
        if name in self.items:
            self.items[name]["qty"] += qty
        else:
            self.items[name] = {"price": price, "qty": qty}
    def remove(self, name):
        if name not in self.items:
            raise KeyError(f"'{name}' مش في العربة")
        del self.items[name]
    @property
    def total(self):
        return sum(i["price"] * i["qty"] for i in self.items.values())
    def apply_coupon(self, percent):
        if not 0 <= percent <= 100:
            raise ValueError("الخصم لازم بين 0 و 100")
        return round(self.total * (1 - percent / 100), 2)
    def __len__(self):
        return len(self.items)
    def __contains__(self, name):
        return name in self.items

إزاي تعرف إن اختباراتك قوية؟ بوّظ الكود عمدًا وشوف الاختبارات هتمسك البوظان ولّا لأ. لو شِلت شرط من الدالة والاختبارات لسه ناجحة — يبقى الشرط ده مش متختبر. أدوات زي mutmut بتعمل ده أوتوماتيك على مشاريع كاملة.

🧪 التحدّي 1 — اختبارات تمسك الطفرات ⭐⭐⭐

الدالة is_strong صحيحة: 8 حروف أو أكتر + رقم + حرف كبير. اكتب 3 دوال اختبار على الأقل (test_… بـ assert) بحيث تنجح على الدالة دي، وتفشل لو حد شال أي شرط منها — أو غيّر >= لـ >.

شوف الحل النموذجي 
def is_strong(password):
    return len(password) >= 8 and any(c.isdigit() for c in password) and any(c.isupper() for c in password)
def test_valid():
    assert is_strong("Secret123")
def test_too_short():
    assert not is_strong("Sec12")
def test_no_digit():
    assert not is_strong("SecretWord")
def test_no_upper():
    assert not is_strong("secret123")
def test_exactly_eight():
    assert is_strong("Secret12")
🧪 التحدّي 2 — TDD: محلّل المُدد ⭐⭐⭐

الاختبارات مكتوبة قبل الكود (زي مشروع TDD). اكتب parse_duration(text) اللي بتحوّل "1h30m" لـ 90 دقيقة. بتقبل ساعات بس، دقايق بس، أو الاتنين، وحروف كبيرة ومسافات. أي حاجة تانية (نص فاضي، رقم من غير وحدة، كلام) ← ValueError.

شوف الحل النموذجي 
import re
def parse_duration(text):
    """'1h30m' ← 90"""
    clean = text.replace(" ", "").lower()
    match = re.fullmatch(r"(?:(\d+)h)?(?:(\d+)m)?", clean)
    if not clean or not match:
        raise ValueError(f"صيغة غير مفهومة: {text!r}")
    hours, minutes = match.groups()
    return int(hours or 0) * 60 + int(minutes or 0)

⭐⭐⭐تمرين 5 — TDD من الصفر

بأسلوب TDD، اعمل دالة slugify(text) بتحوّل "Hello World! 123" لـ "hello-world-123": حروف صغيرة، المسافات شرطات، شيل الرموز، من غير شرطات متكرّرة أو في الأطراف. اكتب الاختبارات الأول.

شوف الحل 
import pytest
from slugify import slugify
@pytest.mark.parametrize("text, expected", [
    ("Hello World", "hello-world"),
    ("Hello World! 123", "hello-world-123"),
    ("  spaced  out  ", "spaced-out"),
    ("MiXeD CaSe", "mixed-case"),
    ("multiple---dashes", "multiple-dashes"),
    ("!!!", ""),
    ("", ""),
    ("a", "a"),
    ("Python 3.12 Rocks!", "python-3-12-rocks"),
])
def test_slugify(text, expected):
    assert slugify(text) == expected
def test_non_string_raises():
    with pytest.raises(TypeError):
        slugify(123)

import re
def slugify(text):
    if not isinstance(text, str):
        raise TypeError("المدخل لازم يكون نص")
    text = text.lower()
    text = re.sub(r"[^a-z0-9]+", "-", text)     # أي حاجة مش حرف/رقم → شرطة
    return text.strip("-")                       # شيل الشرطات من الأطراف

3 سطور منطق، و10 اختبارات. لاحظ إن الاختبارات هي اللي اكتشفت الحالات الحدّية ("!!!" والنص الفاضي).

⭐⭐⭐تمرين 6 — محاكاة

عندك def get_exchange_rate(): ... بتجيب سعر الدولار من API. اكتب اختبار بيحاكي الـ API ويتأكّد إن الدالة بترجّع الرقم الصح، واختبار تاني للفشل.

شوف الحل 
import pytest
from unittest.mock import patch, Mock
from currency import get_exchange_rate
@patch("currency.requests.get")
def test_returns_rate(mock_get):
    fake = Mock()
    fake.json.return_value = {"rates": {"EGP": 48.25}}
    fake.raise_for_status.return_value = None
    mock_get.return_value = fake
    assert get_exchange_rate() == 48.25
    mock_get.assert_called_once()
@patch("currency.requests.get")
def test_network_error_raises(mock_get):
    mock_get.side_effect = ConnectionError("مفيش نت")
    with pytest.raises(ConnectionError):
        get_exchange_rate()
@patch("currency.requests.get")
def test_missing_key_raises(mock_get):
    fake = Mock()
    fake.json.return_value = {"rates": {}}        # الرد ناقص
    fake.raise_for_status.return_value = None
    mock_get.return_value = fake
    with pytest.raises(KeyError):
        get_exchange_rate()

side_effect بتخلّي الدالة الوهمية ترمي خطأ بدل ما ترجّع قيمة — فبنقدر نختبر سلوك برنامجنا لما النت يفصل، من غير ما نفصل النت فعلًا.
