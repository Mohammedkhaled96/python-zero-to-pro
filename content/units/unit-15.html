البرمجة الكائنية (OOP): اعمل أنواعك بنفسك

من الوحدة الأولى وإنت شايف كلمة class في ناتج type()، ومن درس دوال القوائم وإنت شايف __len__ و__iter__. الوحدة دي بتفكّ اللغز: كل حاجة في بايثون كائن، ودلوقتي هتعرف تعمل كائناتك إنت.

2. __str__ و __repr__

class Money:
    def __init__(self, amount, currency="EGP"):
        self.amount = amount
        self.currency = currency
    def __str__(self):
        """للمستخدم النهائي — بتتنادى من print() و str()"""
        return f"{self.amount:,.2f} {self.currency}"
    def __repr__(self):
        """للمبرمج — بتتنادى في الـ Terminal وجوّه القوائم"""
        return f"Money({self.amount!r}, {self.currency!r})"
m = Money(1500.5)
print(m)              # 1,500.50 EGP      ← استخدم __str__
print(str(m))         # 1,500.50 EGP
print(repr(m))        # Money(1500.5, 'EGP')
print([m])            # [Money(1500.5, 'EGP')]  ← القوائم بتستخدم __repr__

class Money:

        self.currency = currency

        if not isinstance(other, Money):

4. __len__ و __contains__ و __getitem__

class Playlist:
    def __init__(self, name):
        self.name = name
        self.songs = []
    def add(self, song):
        self.songs.append(song)
        return self                      # يسمح بالتسلسل
    def __len__(self):
        return len(self.songs)
    def __contains__(self, song):
        return song in self.songs
    def __getitem__(self, index):
        return self.songs[index]
    def __str__(self):
        return f"{self.name} ({len(self)} أغنية)"

class Money:
    def __init__(self, amount):
        self.amount = amount
    def __add__(self, other):
        return Money(self.amount + other.amount)
    def __sub__(self, other):
        return Money(self.amount - other.amount)
    def __mul__(self, factor):
        return Money(self.amount * factor)
    def __lt__(self, other):
        return self.amount < other.amount
    def __str__(self):
        return f"{self.amount} EGP"

b = Money(30)

🧪 التحدّي 1 — الحساب البنكي ⭐⭐⭐

كلاس BankAccount(owner, balance=0): deposit(amount) و withdraw(amount) بيرموا ValueError لأي مبلغ ≤ 0، والسحب أكبر من الرصيد كمان ValueError. balance خاصية للقراءة بس (@property)، و str(acc) ترجّع "Account(sara: 150)".

شوف الحل النموذجي 
class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self._balance = balance
    @property
    def balance(self):
        return self._balance
    @staticmethod
    def _check(amount):
        if amount <= 0:
            raise ValueError("المبلغ لازم يكون موجب")
    def deposit(self, amount):
        self._check(amount)
        self._balance += amount
    def withdraw(self, amount):
        self._check(amount)
        if amount > self._balance:
            raise ValueError("الرصيد مش كفاية")
        self._balance -= amount
    def __str__(self):
        return f"Account({self.owner}: {self._balance})"

        self.w, self.h = w, h

🧪 التحدّي 4 — الطلب ودورة حياته (dataclass + Enum) ⭐⭐⭐

كمّل Order كـ dataclass: customer، و items (قائمة Item فاضية افتراضيًا ومستقلّة لكل طلب)، و status (افتراضيًا OrderStatus.NEW). وضيف: total (property)، و add(item) (مسموحة في NEW بس)، و pay() (من NEW لـ PAID)، و ship() (من PAID لـ SHIPPED). أي انتقال غلط ← ValueError.

شوف الحل النموذجي 
from dataclasses import dataclass, field
from enum import Enum
class OrderStatus(Enum):
    NEW = "new"
    PAID = "paid"
    SHIPPED = "shipped"
@dataclass
class Item:
    name: str
    price: float
    qty: int = 1
@dataclass
class Order:
    customer: str
    items: list[Item] = field(default_factory=list)
    status: OrderStatus = OrderStatus.NEW
    @property
    def total(self):
        return sum(i.price * i.qty for i in self.items)
    def _require(self, expected, action):
        if self.status is not expected:
            raise ValueError(f"مينفعش {action} والطلب {self.status.value}")
    def add(self, item):
        self._require(OrderStatus.NEW, "تضيف بنود")
        self.items.append(item)
    def pay(self):
        self._require(OrderStatus.NEW, "تدفع")
        self.status = OrderStatus.PAID
    def ship(self):
        self._require(OrderStatus.PAID, "تشحن")
        self.status = OrderStatus.SHIPPED

⭐⭐تمرين 3 — كسر عادي بالدوال السحرية

from math import gcd
class Fraction:
    def __init__(self, num, den):
        if den == 0:
            raise ValueError("المقام ما ينفعش يكون صفر")
        g = gcd(num, den)
        self.num = num // g
        self.den = den // g
    def __add__(self, other):
        return Fraction(self.num * other.den + other.num * self.den,
                        self.den * other.den)
    def __mul__(self, other):
        return Fraction(self.num * other.num, self.den * other.den)
    def __eq__(self, other):
        return self.num == other.num and self.den == other.den
    def __str__(self):
        return f"{self.num}/{self.den}"
print(Fraction(1, 2) + Fraction(1, 3))      # 5/6
print(Fraction(2, 4) == Fraction(1, 2))     # True — الاختصار خلّاهم متساويين
print(Fraction(2, 3) * Fraction(3, 4))      # 1/2

⭐⭐⭐تمرين 6 — @property مع تحقّق

اعمل كلاس Rectangle بخاصيتين width وheight ما بيقبلوش قيم سالبة، وخاصية محسوبة area للقراءة بس.

شوف الحل 
class Rectangle:
    def __init__(self, width, height):
        self.width = width
        self.height = height
    @property
    def width(self):
        return self._width
    @width.setter
    def width(self, value):
        if value <= 0:
            raise ValueError("العرض لازم يكون موجب")
        self._width = value
    @property
    def height(self):
        return self._height
    @height.setter
    def height(self, value):
        if value <= 0:
            raise ValueError("الطول لازم يكون موجب")
        self._height = value
    @property
    def area(self):
        return self._width * self._height

print(r.area)            # 12
r.width = 10
print(r.area)            # 40
r.width = -5             # ❌ ValueError
