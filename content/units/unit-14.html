مكتبة requests والـ APIs درس 73

الهدف: تجيب بيانات حيّة من الإنترنت — أكتر حاجة هتبهرك في بايثون.

1. إيه هو الـ API؟

الـ API (Application Programming Interface) هو «شبّاك خدمة» على الإنترنت. إنت بتبعت طلب على رابط معيّن، والخادم بيرجّع لك بيانات منظّمة (غالبًا بصيغة JSON) بدل صفحة ويب. 

  برنامجك ──── GET request ───►  خادم الـ API
                                      │
  برنامجك ◄─── JSON response ─────────┘

  {"rates": {"EGP": 48.5, "EUR": 0.92}}
دورة طلب API

وJSON شكله زي القاموس في بايثون بالظبط: {"key": "value"}. عشان كده بايثون بتحوّله لقاموس بسطر واحد.

2. أول طلب

import requests
url = "https://api.exchangerate-api.com/v4/latest/USD"
try:
    response = requests.get(url, timeout=10)
    response.raise_for_status()      # بترمي خطأ لو الرد فيه مشكلة
    data = response.json()           # بتحوّل JSON لقاموس بايثون
    egp_rate = data["rates"]["EGP"]
    print("1 USD = {} EGP".format(egp_rate))
except requests.RequestException as e:
    print("Error fetching exchange rate:", e)

مليون سبب ممكن يفشل الطلب: النت مفصول، الخادم واقع، الرابط اتغيّر، مهلة انتهت. ما تكتبش requests.get() من غير try/except أبدًا. وضيف timeout=10 عشان البرنامج ما يعلّقش للأبد.

4. تطبيق كامل: تطبيق الطقس

import requests
def fetch_weather_data():
    api_url = "https://api.open-meteo.com/v1/forecast"
    params = {
        "latitude": 30.0444,
        "longitude": 31.2357,
        "current_weather": "true"
    }
    try:
        response = requests.get(api_url, params=params, timeout=10)
        response.raise_for_status()
        return response.json()
    except requests.RequestException as e:
        raise Exception("Error fetching weather data: {}".format(e))
def map_weather_code_to_status(code):
    weather_status_map = {
        0: "Clear sky", 1: "Mainly clear", 2: "Partly cloudy", 3: "Overcast",
        45: "Fog", 48: "Depositing rime fog",
        51: "Light drizzle", 53: "Moderate drizzle", 55: "Dense drizzle",
        61: "Slight rain", 63: "Moderate rain", 65: "Heavy rain",
        71: "Slight snowfall", 73: "Moderate snowfall", 75: "Heavy snowfall",
        80: "Slight rain showers", 81: "Moderate rain showers", 82: "Violent rain showers",
        95: "Thunderstorm", 96: "Thunderstorm with slight hail", 99: "Thunderstorm with heavy hail"
    }
    return weather_status_map.get(code, "Unknown weather condition")
def parse_weather_data(data):
    if "current_weather" not in data:
        raise Exception("Missing 'current_weather' in API response.")
    weather = data["current_weather"]
    temperature = weather.get("temperature")
    weather_code = weather.get("weathercode")
    if temperature is None or weather_code is None:
        raise Exception("Incomplete weather data in the API response.")
    return temperature, map_weather_code_to_status(weather_code)
def main():
    try:
        data = fetch_weather_data()
        temperature, status = parse_weather_data(data)
        print("Current temperature in Cairo, Egypt: {}°C".format(temperature))
        print("Weather status: {}".format(status))
    except Exception as e:
        print("An error occurred:", e)
if __name__ == "__main__":
    main()

5. ليه الكود ده مكتوب كويس؟

  • كل دالة ليها مهمّة واحدة: واحدة بتجيب، واحدة بتترجم الكود، واحدة بتحلّل، وواحدة بتنسّق. لو الـ API اتغيّر، هتعدّل في دالة واحدة بس.
  • params كقاموس: مكتبة requests بتبني الرابط بنفسها. أنضف وأأمن من دمج النصوص يدويًا.
  • .get() بدل []: من الوحدة 10 — بتحمي من KeyError لو الـ API رجّع بيانات ناقصة.
  • raise برسالة واضحة: بدل ما نكمّل ببيانات فاسدة، بنوقف ونقول المشكلة إيه.
  • weather_status_map.get(code, "Unknown"): قيمة افتراضية لو الكود مش في الجدول.

import requests
def fetch(url):
    try:
        response = requests.get(url, timeout=3)
        response.raise_for_status()
        return response.json()
    except requests.Timeout:
        print("⏱ الخادم اتأخّر — جرّب تاني بعدين")
    except requests.ConnectionError:
        print("🌐 مفيش اتصال بالخادم")

        print(f"❌ الخادم رد بخطأ {e.response.status_code}")
    except ValueError:
        print("📄 الرد مش JSON")
    return None

10. إعادة المحاولة واحترام حدود الخدمة

import time
import requests
def get_with_retry(url, attempts=3, wait=1):
    for attempt in range(1, attempts + 1):
        try:
            response = requests.get(url, timeout=5)
            response.raise_for_status()
            return response
        except (requests.Timeout, requests.ConnectionError, requests.HTTPError) as e:
            print(f"محاولة {attempt} فشلت: {type(e).__name__}")
            if attempt == attempts:
                raise
            time.sleep(wait * attempt)
try:
    get_with_retry("https://httpbin.org/status/503", attempts=2, wait=0.5)
except requests.HTTPError:
    print("استسلمنا بعد كل المحاولات")

💡 قبل ما تستخدم أي API
  • اقرا التوثيق بتاعه: الروابط، المعاملات، شكل الرد، وحدود الطلبات (rate limit).
  • التقسيم لصفحات (pagination): النتايج الكتير بتيجي على دفعات بمعامل زي page — لفّ لحد ما الرد يرجع فاضي.

  • كل الأمثلة هنا متشغّلة على httpbin.org — خدمة مجانية للتجربة بترجّع اللي بتبعته.

1. pandas — التعامل مع الجداول و Excel

import pandas as pd
import os
def append_to_excel(name, phone, file_path='output.xlsx'):
    new_data = pd.DataFrame({'name': [name], 'phone': [phone]})
    if os.path.exists(file_path):
        existing_data = pd.read_excel(file_path)
        updated_data = pd.concat([existing_data, new_data], ignore_index=True)
    else:
        updated_data = new_data
    updated_data.to_excel(file_path, index=False, engine='openpyxl')
    print("Data is saved successfully.")
    print(pd.read_excel(file_path))
input_name = input("Enter name: ")
input_phone = input("Enter phone: ")
append_to_excel(input_name, input_phone)

import pandas as pd
def search_excel(search_value, file_path="output.xlsx"):
    df = pd.read_excel(file_path)
    result = df[df.apply(
        lambda row: row.astype(str).str.contains(search_value, na=False, case=False).any(),
        axis=1
    )]
    return result
search_value = input("Enter the search term: ")
try:
    results = search_excel(search_value)
    if not results.empty:
        print("Search Results:")
        print(results)
    else:
        print("No results found.")
except Exception as e:
    print(f"An error occurred: {e}")

from translate import Translator
def translate_text():
    text_to_translate = input("Enter the text to translate: ")
    target_language = input("Enter the target language (default is English): ").lower() or 'en'
    translator = Translator(to_lang=target_language)
    translation = translator.translate(text_to_translate)
    print(f'Original text: {text_to_translate}')
    print(f'Translated text: {translation}')

import pytesseract
from pdf2image import convert_from_path
import numpy as np
def pdf_to_text_tesseract(pdf_path="file.pdf", output_txt_path="output.txt"):
    try:
        images = convert_from_path(pdf_path)
        print(f"Total pages to process: {len(images)}")
        with open(output_txt_path, 'w', encoding='utf-8') as file:
            for image in images:
                np_image = np.array(image)
                text = pytesseract.image_to_string(np_image, lang='ara')
                file.write(text + "\n\n--- Page Break ---\n\n")
        print(f"OCR complete. Output saved to {output_txt_path}")
    except Exception as e:
        print(f"An error occurred: {e}")

4. التشغيل مع بعض: gather

هنا بتحصل الفايدة: التلات طلبات بيبدأوا في نفس اللحظة، والوقت الكلي = أطول طلب مش مجموعهم.

import asyncio
import time
async def fetch(name, seconds):
    await asyncio.sleep(seconds)
    return f"{name} ({seconds}s)"
async def main():
    start = time.perf_counter()
    results = await asyncio.gather(
        fetch("طلب 1", 0.3),
        fetch("طلب 2", 0.2),
        fetch("طلب 3", 0.1),
    )
    print(results)
    print("الوقت:", round(time.perf_counter() - start, 1), "ثانية")
asyncio.run(main())

5. مهام، ومهلة، وأول نتيجة تخلص

import asyncio
async def work(name, seconds):
    await asyncio.sleep(seconds)
    print("خلص:", name)
    return name
async def main():
    task = asyncio.create_task(work("خلفية", 0.1))
    print("المهمة اتجدولت وإحنا لسه شغالين")
    await task
    try:
        await asyncio.wait_for(work("بطيء", 1.0), timeout=0.2)
    except asyncio.TimeoutError:
        print("اتأخّر فألغيناه")
    done, pending = await asyncio.wait(
        [asyncio.create_task(work("سريع", 0.1)), asyncio.create_task(work("متأخّر", 0.5))],
        return_when=asyncio.FIRST_COMPLETED,
    )
    print("خلصت:", len(done), "· فاضل:", len(pending))
    for t in pending:
        t.cancel()
asyncio.run(main())

7. الحسابات التقيلة: العمليات مش الخيوط

from concurrent.futures import ProcessPoolExecutor
import time
def heavy(n):
    return sum(i * i for i in range(n))
if __name__ == "__main__":
    numbers = [2_000_000] * 4
    start = time.perf_counter()
    serial = [heavy(n) for n in numbers]
    print("بالتوالي:", round(time.perf_counter() - start, 2), "ثانية")
    start = time.perf_counter()
    with ProcessPoolExecutor() as pool:
        parallel = list(pool.map(heavy, numbers))
    print("بالتوازي:", round(time.perf_counter() - start, 2), "ثانية")
    print(serial == parallel)

🧪 التحدّي 1 — ملخّص الطقس ⭐⭐

summarize(json_text) بتاخد نص JSON شكله {"name": "Cairo", "main": {"temp": 31.4}, "weather": [{"description": "clear sky"}]} وترجّع "Cairo: 31°C — clear sky" (الحرارة مقرّبة). لو فيه مفتاح ناقص ترجّع "بيانات ناقصة"، ولو النص مش JSON أصلًا ترجّع "رد غير صالح".

شوف الحل النموذجي 
import json
def summarize(json_text):
    try:
        data = json.loads(json_text)
    except json.JSONDecodeError:
        return "رد غير صالح"
    try:
        city = data["name"]
        temp = round(data["main"]["temp"])
        description = data["weather"][0]["description"]
    except (KeyError, IndexError, TypeError):
        return "بيانات ناقصة"
    return f"{city}: {temp}°C — {description}"

import re
EMAIL = re.compile(r"[\w.+-]+@[\w-]+(?:\.[\w-]+)+")
def extract_emails(text):
    found = {match.lower() for match in EMAIL.findall(text)}
    return sorted(found)
